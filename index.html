Sample SQL Logic

Visible technical evidence without exposing sensitive data.

SELECT
    e.equipment_id,
    e.equipment_name,
    ROUND(SUM(u.idle_hours) / NULLIF(SUM(u.hours_used + u.idle_hours), 0), 2) AS idle_ratio,
    SUM(u.hours_used) AS total_hours_used,
    SUM(u.idle_hours) AS total_idle_hours
FROM equipment e
JOIN equipment_usage_logs u
    ON e.equipment_id = u.equipment_id
GROUP BY e.equipment_id, e.equipment_name
HAVING idle_ratio >= 0.30
ORDER BY idle_ratio DESC;
